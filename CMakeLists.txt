cmake_minimum_required(VERSION 3.16)
project(stm32_sent_decoder C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SENT_ENABLE_WERROR "Treat compiler warnings as errors" ON)

add_library(sent_decoder src/sent_decoder.c)
target_include_directories(sent_decoder PUBLIC include)
target_compile_options(sent_decoder PRIVATE -Wall -Wextra -Wpedantic)
if(SENT_ENABLE_WERROR)
  target_compile_options(sent_decoder PRIVATE -Werror)
endif()

add_library(sent_simulator simulator/sent_sim.c)
target_include_directories(sent_simulator PUBLIC include simulator)
target_link_libraries(sent_simulator PUBLIC sent_decoder)
target_compile_options(sent_simulator PRIVATE -Wall -Wextra -Wpedantic)
if(SENT_ENABLE_WERROR)
  target_compile_options(sent_simulator PRIVATE -Werror)
endif()

add_executable(sent_demo app/main.c)
target_link_libraries(sent_demo PRIVATE sent_decoder sent_simulator)
target_compile_options(sent_demo PRIVATE -Wall -Wextra -Wpedantic)
if(SENT_ENABLE_WERROR)
  target_compile_options(sent_demo PRIVATE -Werror)
endif()

enable_testing()
add_executable(test_sent_decoder tests/test_sent_decoder.c)
target_link_libraries(test_sent_decoder PRIVATE sent_decoder sent_simulator)
target_compile_options(test_sent_decoder PRIVATE -Wall -Wextra -Wpedantic)
if(SENT_ENABLE_WERROR)
  target_compile_options(test_sent_decoder PRIVATE -Werror)
endif()
add_test(NAME sent_decoder_unit_tests COMMAND test_sent_decoder)
